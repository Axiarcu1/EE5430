\section{Bài 1}

\subsection{Đề bài}
Một đường truyền sóng $50~\Omega$ nối với tải $Z_L = 25 - j50~\Omega$, bước sóng trên đường truyền $\lambda = 10$~cm. Dùng đồ thị Smith tìm hệ số phản xạ tại tải, hệ số sóng đứng VSWR, vị trí điện áp cực tiểu và cực đại đầu tiên tính từ tải, và trở kháng đường dây tại hai vị trí đó.

\subsection{Lời giải}

\subsubsection{Hệ số phản xạ tại tải}

Trở kháng chuẩn hóa:
\[
z_L = \frac{Z_L}{Z_0} = \frac{25 - j50}{50} = 0{,}5 - j1
\]

Hệ số phản xạ tại tải:
\[
\Gamma_L = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{25 - j50 - 50}{25 - j50 + 50} = \frac{-25 - j50}{75 - j50}
\]

Nhân liên hợp:
\[
\Gamma_L = \frac{(-25 - j50)(75 + j50)}{(75)^2 + (50)^2} = \frac{-1875 - j1250 - j3750 - j^2 2500}{5625 + 2500}
\]
\[
= \frac{-1875 + 2500 - j(1250 + 3750)}{8125} = \frac{625 - j5000}{8125}
\]
\[
\Gamma_L = 0{,}0769 - j0{,}6154
\]

Biên độ:
\[
|\Gamma_L| = \sqrt{0{,}0769^2 + 0{,}6154^2} = \sqrt{0{,}00592 + 0{,}37871} = \sqrt{0{,}38462} = 0{,}6202
\]

Góc pha:
\[
\theta_\Gamma = \arctan\left(\frac{-0{,}6154}{0{,}0769}\right) = \arctan(-8{,}003) = -82{,}87^\circ
\]

Do phần thực dương và phần ảo âm (góc phần tư thứ IV):
\[
\boxed{\Gamma_L = 0{,}6202 \angle -82{,}87^\circ}
\]

\subsubsection{Hệ số sóng đứng VSWR}

\[
\text{VSWR} = \frac{1 + |\Gamma_L|}{1 - |\Gamma_L|} = \frac{1 + 0{,}6202}{1 - 0{,}6202} = \frac{1{,}6202}{0{,}3798}
\]
\[
\boxed{\text{VSWR} = 4{,}266}
\]

\subsubsection{Vị trí điện áp cực đại và cực tiểu đầu tiên}

Điện áp cực đại xảy ra khi pha tổng $\theta_\Gamma - 2\beta d = 0$ (hoặc bội $2\pi$):
\[
d_{\max} = \frac{\theta_\Gamma}{2\beta} = \frac{\theta_\Gamma}{2 \times 2\pi/\lambda} = \frac{\theta_\Gamma \lambda}{4\pi}
\]

Với $\theta_\Gamma = -82{,}87^\circ = -1{,}4464$~rad:
\[
d_{\max} = \frac{-1{,}4464 \times 10}{4\pi} = \frac{-14{,}464}{12{,}566} = -1{,}151~\text{cm}
\]

Vì $d_{\max} < 0$, cộng thêm $\lambda/2$:
\[
d_{\max} = -1{,}151 + 5 = \boxed{3{,}849~\text{cm}}
\]

Điện áp cực tiểu cách cực đại $\lambda/4$:
\[
d_{\min} = d_{\max} - \frac{\lambda}{4} = 3{,}849 - 2{,}5 = 1{,}349~\text{cm}
\]

Vì $\theta_\Gamma < 0$ nên $V_{\min}$ gần tải hơn $V_{\max}$:
\[
\boxed{d_{\min} = 1{,}349~\text{cm}}
\]

\subsubsection{Trở kháng tại vị trí $V_{\max}$ và $V_{\min}$}

\textbf{Tại $V_{\max}$:} Trở kháng thuần trở và bằng:
\[
Z(d_{\max}) = Z_0 \times \text{VSWR} = 50 \times 4{,}266 = \boxed{213{,}3~\Omega}
\]

\textbf{Tại $V_{\min}$:} Trở kháng thuần trở và bằng:
\[
Z(d_{\min}) = \frac{Z_0}{\text{VSWR}} = \frac{50}{4{,}266} = \boxed{11{,}72~\Omega}
\]

\subsubsection{Kiểm tra bằng đồ thị Smith}

Trên đồ thị Smith:
\begin{enumerate}
    \item Xác định điểm $z_L = 0{,}5 - j1$ trên đồ thị.
    \item Vẽ đường tròn $|\Gamma| = \text{const}$ qua điểm $z_L$ (bán kính $|\Gamma_L| = 0{,}62$).
    \item $V_{\max}$ ứng với giao điểm đường tròn với trục thực dương (phải): $z = \text{VSWR} = 4{,}27$.
    \item $V_{\min}$ ứng với giao điểm đường tròn với trục thực dương (trái): $z = 1/\text{VSWR} = 0{,}234$.
    \item Đọc khoảng cách trên thang ``wavelengths toward generator'' từ $z_L$ đến $V_{\max}$: $d_{\max}/\lambda \approx 0{,}385$.
    \item Đọc khoảng cách từ $z_L$ đến $V_{\min}$: $d_{\min}/\lambda \approx 0{,}135$.
\end{enumerate}

Kết quả tính giải tích phù hợp với đồ thị Smith.

\section{Bài 2}

\subsection{Đề bài}
Một đường truyền sóng điện môi không khí có trở kháng đặc tính $75~\Omega$, tần số làm việc $1{,}5$~GHz, nối với tải $Z_L = 150 + j75~\Omega$. Tìm VSWR, vị trí $V_{\max}$ và $V_{\min}$ đầu tiên tính từ tải, và trở kháng nhìn vào đường truyền tại điểm cách tải $3$~cm.

\subsection{Lời giải}

\subsubsection{Bước sóng và hằng số pha}

Vì điện môi không khí, vận tốc truyền sóng $v_p = c = 3 \times 10^8$~m/s:
\[
\lambda = \frac{v_p}{f} = \frac{3 \times 10^8}{1{,}5 \times 10^9} = 0{,}2~\text{m} = 20~\text{cm}
\]
\[
\beta = \frac{2\pi}{\lambda} = \frac{2\pi}{0{,}2} = 10\pi~\text{rad/m}
\]

\subsubsection{Hệ số phản xạ tại tải}

\[
\Gamma_L = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{150 + j75 - 75}{150 + j75 + 75} = \frac{75 + j75}{225 + j75}
\]

Nhân liên hợp:
\[
\Gamma_L = \frac{(75 + j75)(225 - j75)}{225^2 + 75^2} = \frac{16875 - j5625 + j16875 - j^2 5625}{50625 + 5625}
\]
\[
= \frac{16875 + 5625 + j(16875 - 5625)}{56250} = \frac{22500 + j11250}{56250}
\]
\[
\Gamma_L = 0{,}4 + j0{,}2
\]

\[
|\Gamma_L| = \sqrt{0{,}4^2 + 0{,}2^2} = \sqrt{0{,}16 + 0{,}04} = \sqrt{0{,}20} = 0{,}4472
\]

\[
\theta_\Gamma = \arctan\left(\frac{0{,}2}{0{,}4}\right) = \arctan(0{,}5) = 26{,}57^\circ
\]

\[
\Gamma_L = 0{,}4472\angle 26{,}57^\circ
\]

\subsubsection{VSWR}

\[
\boxed{\text{VSWR} = \frac{1 + 0{,}4472}{1 - 0{,}4472} = \frac{1{,}4472}{0{,}5528} = 2{,}618}
\]

\subsubsection{Vị trí $V_{\max}$ và $V_{\min}$ đầu tiên}

\[
d_{\max} = \frac{\theta_\Gamma \lambda}{4\pi} = \frac{26{,}57^\circ \times \pi/180^\circ \times 20}{4\pi} = \frac{0{,}4637 \times 20}{4\pi} = \frac{9{,}274}{12{,}566}
\]
\[
\boxed{d_{\max} = 0{,}738~\text{cm}}
\]

\[
d_{\min} = d_{\max} + \frac{\lambda}{4} = 0{,}738 + 5 = \boxed{5{,}738~\text{cm}}
\]

(Vì $\theta_\Gamma > 0$, $V_{\max}$ gần tải hơn.)

\subsubsection{Trở kháng tại $d = 3$~cm}

\[
\beta d = 10\pi \times 0{,}03 = 0{,}3\pi~\text{rad} = 54^\circ
\]

\[
Z_{in} = Z_0 \frac{Z_L + jZ_0\tan(\beta d)}{Z_0 + jZ_L\tan(\beta d)}
\]

$\tan(54^\circ) = 1{,}3764$

Tử:
\[
Z_L + jZ_0\tan(\beta d) = (150 + j75) + j75 \times 1{,}3764
\]
\[
= 150 + j75 + j103{,}23 = 150 + j178{,}23
\]

Mẫu:
\[
Z_0 + jZ_L\tan(\beta d) = 75 + j(150 + j75) \times 1{,}3764
\]
\[
= 75 + j150 \times 1{,}3764 + j^2 75 \times 1{,}3764
\]
\[
= 75 - 103{,}23 + j206{,}46 = -28{,}23 + j206{,}46
\]

\[
Z_{in} = 75 \times \frac{150 + j178{,}23}{-28{,}23 + j206{,}46}
\]

Nhân liên hợp mẫu:
\[
\frac{(150 + j178{,}23)(-28{,}23 - j206{,}46)}{(-28{,}23)^2 + (206{,}46)^2}
\]

Tử số:
\begin{align*}
&= 150 \times (-28{,}23) + 150 \times (-j206{,}46) + j178{,}23 \times (-28{,}23) + j178{,}23 \times (-j206{,}46) \\
&= -4234{,}5 - j30969 - j5029{,}67 + 36789{,}04 \\
&= (-4234{,}5 + 36789{,}04) + j(-30969 - 5029{,}67) \\
&= 32554{,}54 - j35998{,}67
\end{align*}

Mẫu số:
\[
796{,}93 + 42625{,}73 = 43422{,}66
\]

\[
\frac{Z_{in}}{75} = \frac{32554{,}54 - j35998{,}67}{43422{,}66} = 0{,}7498 - j0{,}8291
\]

\[
\boxed{Z_{in} = 75(0{,}7498 - j0{,}8291) = 56{,}24 - j62{,}18~\Omega}
\]

\section{Bài 3}

\subsection{Đề bài}
Một đường truyền sóng $50~\Omega$ nối với tải $Z_L$ chưa biết có VSWR $= 3$ và $\Delta x_{\min} = 20$~cm (khoảng cách hai điểm cực tiểu liên tiếp). Khi thay tải bằng ngắn mạch, vị trí điểm cực tiểu dịch chuyển $8$~cm về phía nguồn. Tìm $Z_L$.

\subsection{Lời giải}

\subsubsection{Xác định bước sóng}

Khoảng cách giữa hai cực tiểu liên tiếp:
\[
\Delta x_{\min} = \frac{\lambda}{2} \implies \lambda = 2 \times 20 = 40~\text{cm}
\]

\subsubsection{Phân tích phép đo}

Khi nối ngắn mạch, điểm cực tiểu (nút sóng) nằm chính xác tại vị trí tải (bội $\lambda/2$ từ tải). Khi thay ngắn mạch bằng tải $Z_L$, điểm cực tiểu dịch chuyển.

Gọi $d_{\min}$ là khoảng cách từ tải đến $V_{\min}$ đầu tiên.

Khi ngắn mạch, $V_{\min}$ nằm tại tải ($d = 0$) và lặp lại mỗi $\lambda/2$.

Vị trí cực tiểu khi ngắn mạch dịch về phía nguồn $8$~cm so với khi có tải. Điều này có nghĩa là khi có tải, cực tiểu gần tải hơn (gần nguồn hơn tính từ ngắn mạch):
\[
d_{\min} = 8~\text{cm}
\]

\textit{Lưu ý:} Cực tiểu dịch về phía nguồn nghĩa là cực tiểu khi ngắn mạch nằm gần tải hơn, và khi có $Z_L$ thì cực tiểu xa tải hơn $8$~cm.

\subsubsection{Tính hệ số phản xạ}

Biên độ hệ số phản xạ:
\[
|\Gamma_L| = \frac{\text{VSWR} - 1}{\text{VSWR} + 1} = \frac{3 - 1}{3 + 1} = \frac{2}{4} = 0{,}5
\]

Tại $V_{\min}$, pha tổng: $\theta_\Gamma - 2\beta d_{\min} = -\pi$ (hoặc bội lẻ $\pi$):
\[
\theta_\Gamma = -\pi + 2\beta d_{\min} = -\pi + 2 \times \frac{2\pi}{\lambda} \times d_{\min}
\]
\[
= -\pi + \frac{4\pi \times 8}{40} = -\pi + 0{,}8\pi = -0{,}2\pi~\text{rad} = -36^\circ
\]

Vậy:
\[
\Gamma_L = 0{,}5 \angle (-36^\circ) = 0{,}5(\cos 36^\circ - j\sin 36^\circ) = 0{,}5(0{,}809 - j0{,}588)
\]
\[
\Gamma_L = 0{,}4045 - j0{,}2939
\]

\subsubsection{Tính trở kháng tải}

\[
Z_L = Z_0 \frac{1 + \Gamma_L}{1 - \Gamma_L} = 50 \times \frac{1 + 0{,}4045 - j0{,}2939}{1 - 0{,}4045 + j0{,}2939}
\]
\[
= 50 \times \frac{1{,}4045 - j0{,}2939}{0{,}5955 + j0{,}2939}
\]

Nhân liên hợp:
\[
\frac{(1{,}4045 - j0{,}2939)(0{,}5955 - j0{,}2939)}{0{,}5955^2 + 0{,}2939^2}
\]

Tử:
\begin{align*}
&= 1{,}4045 \times 0{,}5955 - j1{,}4045 \times 0{,}2939 - j0{,}2939 \times 0{,}5955 + j^2 0{,}2939 \times 0{,}2939 \\
&= 0{,}8362 - j0{,}4128 - j0{,}1750 - 0{,}0864 \\
&= 0{,}7498 - j0{,}5878
\end{align*}

Mẫu:
\[
0{,}3546 + 0{,}0864 = 0{,}4410
\]

\[
\frac{Z_L}{50} = \frac{0{,}7498 - j0{,}5878}{0{,}4410} = 1{,}700 - j1{,}333
\]

\[
\boxed{Z_L = 50(1{,}700 - j1{,}333) = 85{,}0 - j66{,}7~\Omega}
\]

\section{Bài 4}

\subsection{Đề bài}
Một đường truyền sóng $100~\Omega$ nối với tải $Z_L$ có VSWR $= 1{,}5$ và $\Delta x_{\min} = 30$~cm. Khi thay tải bằng ngắn mạch, vị trí điểm cực tiểu dịch chuyển $6$~cm về phía tải. Tìm bước sóng và $Z_L$. Tải có tính cảm hay tính dung?

\subsection{Lời giải}

\subsubsection{Bước sóng}

\[
\lambda = 2 \times \Delta x_{\min} = 2 \times 30 = 60~\text{cm}
\]

\subsubsection{Phân tích}

Khi ngắn mạch, cực tiểu dịch $6$~cm về phía tải (so với khi có $Z_L$). Nghĩa là cực tiểu khi có tải nằm xa tải hơn so với khi ngắn mạch.

Điều này tương đương: khi ngắn mạch, cực tiểu tại $d = 0$ (và bội $\lambda/2$). Khi có tải, cực tiểu dịch ra xa tải thêm $6$~cm. Nhưng đề bảo ``dịch về phía tải'', tức cực tiểu khi ngắn mạch xa tải hơn, và khi ngắn mạch nó dịch lại gần tải:

Cực tiểu khi ngắn mạch nằm tại bội $\lambda/2$ từ tải. Gần nhất là $d = 0$. Khi có tải, cực tiểu đầu tiên ở $d_{\min}$.

``Dịch $6$~cm về phía tải'' nghĩa là $d_{\min}$ khi có tải giảm đi, hoặc tương đương: cực tiểu khi ngắn mạch (ở vị trí $n\lambda/2$ nào đó trên thang đo) dịch về phía tải $6$~cm so với cực tiểu khi có $Z_L$.

Cách tiếp cận: Khi ngắn mạch, cực tiểu tại tải. Cực tiểu kế tiếp cách $\lambda/2 = 30$~cm về phía nguồn. Khi thay bằng $Z_L$, cực tiểu gần nhất nằm ở $d_{\min}$.

Vì cực tiểu dịch về phía tải $6$~cm: cực tiểu khi ngắn mạch ở vị trí nào đó, và khi thay lại $Z_L$, cực tiểu nằm xa tải hơn $6$~cm:
\[
d_{\min} = \frac{\lambda}{2} - 6 = 30 - 6 = 24~\text{cm}
\]

\subsubsection{Hệ số phản xạ}

\[
|\Gamma_L| = \frac{1{,}5 - 1}{1{,}5 + 1} = \frac{0{,}5}{2{,}5} = 0{,}2
\]

Pha hệ số phản xạ:
\[
\theta_\Gamma = -\pi + 2\beta d_{\min} = -\pi + \frac{4\pi \times 24}{60} = -\pi + 1{,}6\pi = 0{,}6\pi~\text{rad} = 108^\circ
\]

\[
\Gamma_L = 0{,}2\angle 108^\circ = 0{,}2(\cos 108^\circ + j\sin 108^\circ)
\]
\[
= 0{,}2(-0{,}3090 + j0{,}9511) = -0{,}0618 + j0{,}1902
\]

\subsubsection{Trở kháng tải}

\[
Z_L = Z_0 \frac{1 + \Gamma_L}{1 - \Gamma_L} = 100 \times \frac{1 - 0{,}0618 + j0{,}1902}{1 + 0{,}0618 - j0{,}1902}
\]
\[
= 100 \times \frac{0{,}9382 + j0{,}1902}{1{,}0618 - j0{,}1902}
\]

Nhân liên hợp:
\[
\frac{(0{,}9382 + j0{,}1902)(1{,}0618 + j0{,}1902)}{1{,}0618^2 + 0{,}1902^2}
\]

Tử:
\begin{align*}
&= 0{,}9382 \times 1{,}0618 + j0{,}9382 \times 0{,}1902 + j0{,}1902 \times 1{,}0618 + j^2 0{,}1902^2 \\
&= 0{,}9962 + j0{,}1785 + j0{,}2020 - 0{,}0362 \\
&= 0{,}9600 + j0{,}3805
\end{align*}

Mẫu:
\[
1{,}1274 + 0{,}0362 = 1{,}1636
\]

\[
\frac{Z_L}{100} = \frac{0{,}9600 + j0{,}3805}{1{,}1636} = 0{,}8250 + j0{,}3270
\]

\[
\boxed{Z_L = 100(0{,}8250 + j0{,}3270) = 82{,}5 + j32{,}7~\Omega}
\]

\subsubsection{Tính cảm hay tính dung?}

Phần ảo của $Z_L$ là $+j32{,}7~\Omega > 0$, do đó tải có \textbf{tính cảm (inductive)}.

\[
\boxed{\text{Tải có tính cảm}}
\]

\section{Bài 5}

\subsection{Đề bài}
Một đường truyền sóng điện môi không khí có trở kháng đặc tính $50~\Omega$, đầu cuối nối với tải $Z_L$. Đo sóng đứng được VSWR $= 1{,}8$, hai nút sóng liền tiếp nằm tại vạch $32{,}0$~cm và $44{,}0$~cm của thang đo. Khi nối tắt tải, nút sóng dịch chuyển về phía tải và nằm tại vạch $26{,}6$~cm. Tìm bước sóng, tần số làm việc và trở kháng tải.

\subsection{Lời giải}

\subsubsection{Bước sóng và tần số}

Khoảng cách giữa hai nút sóng liền tiếp (khi có tải):
\[
\frac{\lambda}{2} = 44{,}0 - 32{,}0 = 12{,}0~\text{cm}
\]
\[
\boxed{\lambda = 24{,}0~\text{cm} = 0{,}24~\text{m}}
\]

Tần số (điện môi không khí, $v_p = c$):
\[
\boxed{f = \frac{c}{\lambda} = \frac{3 \times 10^8}{0{,}24} = 1{,}25~\text{GHz}}
\]

\subsubsection{Xác định khoảng cách $d_{\min}$ từ tải đến nút sóng}

Trên thang đo, vạch tăng dần về phía nguồn. Tải nằm ở phía vạch nhỏ.

Khi nối tắt (ngắn mạch), nút sóng nằm chính xác tại vị trí tải và lặp lại mỗi $\lambda/2 = 12$~cm. Nút sóng khi ngắn mạch tại vạch $26{,}6$~cm, nên vị trí tải tương đương cách một bội $\lambda/2$: các nút ngắn mạch tại $26{,}6$~cm, $38{,}6$~cm, $50{,}6$~cm, \ldots

Khi có tải, nút sóng gần nhất nằm tại vạch $32{,}0$~cm.

Khoảng cách từ nút sóng khi ngắn mạch (gần nhất với vạch $32{,}0$~cm) đến nút sóng khi có tải:

Nút ngắn mạch gần $32{,}0$~cm nhất: $26{,}6$~cm hoặc $38{,}6$~cm.

Nút sóng khi có tải dịch từ $26{,}6$ đến $32{,}0$: dịch $5{,}4$~cm về phía nguồn.

\[
\Delta d = 32{,}0 - 26{,}6 = 5{,}4~\text{cm}
\]

Đây là khoảng dịch của cực tiểu về phía nguồn khi thay ngắn mạch bằng tải. Tương đương:
\[
d_{\min} = \Delta d = 5{,}4~\text{cm}
\]

\subsubsection{Hệ số phản xạ}

\[
|\Gamma_L| = \frac{\text{VSWR} - 1}{\text{VSWR} + 1} = \frac{1{,}8 - 1}{1{,}8 + 1} = \frac{0{,}8}{2{,}8} = 0{,}2857
\]

Pha:
\[
\theta_\Gamma = -\pi + 2\beta d_{\min} = -\pi + \frac{4\pi \times 5{,}4}{24} = -\pi + 0{,}9\pi = -0{,}1\pi~\text{rad} = -18^\circ
\]

\[
\Gamma_L = 0{,}2857\angle(-18^\circ) = 0{,}2857(\cos 18^\circ - j\sin 18^\circ)
\]
\[
= 0{,}2857(0{,}9511 - j0{,}3090) = 0{,}2717 - j0{,}0883
\]

\subsubsection{Trở kháng tải}

\[
Z_L = Z_0 \frac{1 + \Gamma_L}{1 - \Gamma_L} = 50 \times \frac{1{,}2717 - j0{,}0883}{0{,}7283 + j0{,}0883}
\]

Nhân liên hợp:
\[
\frac{(1{,}2717 - j0{,}0883)(0{,}7283 - j0{,}0883)}{0{,}7283^2 + 0{,}0883^2}
\]

Tử:
\begin{align*}
&= 1{,}2717 \times 0{,}7283 - j1{,}2717 \times 0{,}0883 - j0{,}0883 \times 0{,}7283 + j^2(0{,}0883)^2 \\
&= 0{,}9262 - j0{,}1123 - j0{,}0643 - 0{,}0078 \\
&= 0{,}9184 - j0{,}1766
\end{align*}

Mẫu:
\[
0{,}5304 + 0{,}0078 = 0{,}5382
\]

\[
\frac{Z_L}{50} = \frac{0{,}9184 - j0{,}1766}{0{,}5382} = 1{,}706 - j0{,}328
\]

\[
\boxed{Z_L = 50(1{,}706 - j0{,}328) = 85{,}3 - j16{,}4~\Omega}
\]

\section{Bài 6}

\subsection{Đề bài}
Một đường truyền sóng điện môi không khí có trở kháng đặc tính $75~\Omega$. Thang đo của khe đo có số chỉ tăng dần về phía nguồn. Với tải $Z_L$, đo được VSWR $= 2{,}5$ và hai nút sóng liền tiếp tại vạch $18{,}2$~cm và $28{,}2$~cm. Khi nối tắt tải, các nút sóng nằm tại vạch $21{,}2$~cm và $31{,}2$~cm. Tìm tần số làm việc và trở kháng tải.

\subsection{Lời giải}

\subsubsection{Bước sóng và tần số}

Khoảng cách giữa hai nút sóng liền tiếp:
\[
\frac{\lambda}{2} = 28{,}2 - 18{,}2 = 10{,}0~\text{cm}
\]
\[
\boxed{\lambda = 20{,}0~\text{cm} = 0{,}20~\text{m}}
\]

Kiểm tra: khi ngắn mạch, $31{,}2 - 21{,}2 = 10{,}0$~cm $= \lambda/2$ \checkmark

Tần số (điện môi không khí):
\[
\boxed{f = \frac{c}{\lambda} = \frac{3 \times 10^8}{0{,}20} = 1{,}5~\text{GHz}}
\]

\subsubsection{Xác định khoảng dịch nút sóng}

Khi nối tắt (ngắn mạch), nút sóng tại: $21{,}2$~cm và $31{,}2$~cm.

Khi có tải, nút sóng tại: $18{,}2$~cm và $28{,}2$~cm.

Nút sóng khi có tải nằm gần tải hơn so với ngắn mạch (vạch nhỏ hơn $\to$ gần tải hơn). Khi ngắn mạch, nút dịch về phía nguồn thêm:
\[
\Delta d = 21{,}2 - 18{,}2 = 3{,}0~\text{cm}
\]

Kiểm tra: $31{,}2 - 28{,}2 = 3{,}0$~cm \checkmark

Nút sóng khi ngắn mạch nằm tại vị trí tải (bội $\lambda/2$). Khi có tải, nút sóng dịch về phía tải $3$~cm. Điều này nghĩa là:

Gọi cực tiểu khi ngắn mạch ở vạch $21{,}2$~cm, và cực tiểu khi có tải gần nhất ở vạch $18{,}2$~cm. Cực tiểu đã dịch $3{,}0$~cm về phía tải (vạch nhỏ hơn).

Khoảng cách từ nút ngắn mạch đến tải: $d = 0$ (mod $\lambda/2$). Cực tiểu khi có tải cách nút ngắn mạch gần nhất:
\[
d_{\min} = \lambda/2 - 3{,}0 = 10{,}0 - 3{,}0 = 7{,}0~\text{cm}
\]

\textit{Giải thích:} Cực tiểu khi có tải nằm trước nút ngắn mạch gần nhất (về phía tải) $3$~cm, tương đương cách nút ngắn mạch trước đó $\lambda/2 - 3 = 7$~cm về phía nguồn.

\subsubsection{Hệ số phản xạ}

\[
|\Gamma_L| = \frac{2{,}5 - 1}{2{,}5 + 1} = \frac{1{,}5}{3{,}5} = 0{,}4286
\]

Pha:
\[
\theta_\Gamma = -\pi + 2\beta d_{\min} = -\pi + \frac{4\pi \times 7{,}0}{20} = -\pi + 1{,}4\pi = 0{,}4\pi~\text{rad} = 72^\circ
\]

\[
\Gamma_L = 0{,}4286\angle 72^\circ = 0{,}4286(\cos 72^\circ + j\sin 72^\circ)
\]
\[
= 0{,}4286(0{,}3090 + j0{,}9511) = 0{,}1324 + j0{,}4076
\]

\subsubsection{Trở kháng tải}

\[
Z_L = Z_0 \frac{1 + \Gamma_L}{1 - \Gamma_L} = 75 \times \frac{1{,}1324 + j0{,}4076}{0{,}8676 - j0{,}4076}
\]

Nhân liên hợp:
\[
\frac{(1{,}1324 + j0{,}4076)(0{,}8676 + j0{,}4076)}{0{,}8676^2 + 0{,}4076^2}
\]

Tử:
\begin{align*}
&= 1{,}1324 \times 0{,}8676 + j1{,}1324 \times 0{,}4076 + j0{,}4076 \times 0{,}8676 + j^2 0{,}4076^2 \\
&= 0{,}9825 + j0{,}4616 + j0{,}3537 - 0{,}1661 \\
&= 0{,}8164 + j0{,}8153
\end{align*}

Mẫu:
\[
0{,}7527 + 0{,}1661 = 0{,}9188
\]

\[
\frac{Z_L}{75} = \frac{0{,}8164 + j0{,}8153}{0{,}9188} = 0{,}8886 + j0{,}8874
\]

\[
\boxed{Z_L = 75(0{,}8886 + j0{,}8874) = 66{,}6 + j66{,}6~\Omega}
\]

\textbf{Kiểm tra:} Tải có phần ảo dương (tính cảm), và $\theta_\Gamma > 0$ cho thấy $V_{\max}$ gần tải, phù hợp với việc nút sóng (cực tiểu) dịch về phía tải so với ngắn mạch.
