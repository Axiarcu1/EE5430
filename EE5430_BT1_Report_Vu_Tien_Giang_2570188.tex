\section*{Tổng quan lý thuyết: Tham số tán xạ và mạng vi ba}
\addcontentsline{toc}{section}{Tổng quan lý thuyết: Tham số tán xạ và mạng vi ba}

Trong dải tần số siêu cao tần (RF/Microwave), việc đo lường trực tiếp điện áp và dòng điện toàn phần tại các cửa của mạng hai cửa hay đa cửa trở nên rất khó khăn do sự xuất hiện của sóng đứng và hiệu ứng phân bố. Thay vào đó, mạng siêu cao tần thường được mô tả thông qua sóng điện áp tới ($a_n$) và sóng điện áp phản xạ ($b_n$), định nghĩa qua ma trận tán xạ $[S]$ (Scattering Matrix):
\[
\mathbf{b} = [S]\mathbf{a} \iff \begin{bmatrix} b_1 \\ b_2 \\ \vdots \\ b_N \end{bmatrix} = \begin{bmatrix} S_{11} & S_{12} & \cdots & S_{1N} \\ S_{21} & S_{22} & \cdots & S_{2N} \\ \vdots & \vdots & \ddots & \vdots \\ S_{N1} & S_{N2} & \cdots & S_{NN} \end{bmatrix} \begin{bmatrix} a_1 \\ a_2 \\ \vdots \\ a_N \end{bmatrix}
\]
với $a_n = \frac{V_n^+}{\sqrt{Z_{0n}}}$ và $b_n = \frac{V_n^-}{\sqrt{Z_{0n}}}$.

\paragraph{Các tính chất cơ bản của ma trận $[S]$:}
\begin{enumerate}
    \item \textbf{Phối hợp trở kháng (Matched):} Cửa thứ $i$ được phối hợp hoàn hảo khi không có sóng phản xạ từ cửa đó nếu tất cả các cửa còn lại được kết cuối phối hợp, tức $S_{ii} = 0$.
    \item \textbf{Tính thuận nghịch (Reciprocal):} Mạng chỉ chứa các môi trường đẳng hướng và linh kiện thụ động tuyến tính thì $[S]$ là ma trận đối xứng:
    \[
    [S] = [S]^T \iff S_{ij} = S_{ji} \quad \forall i, j.
    \]
    \item \textbf{Tính không tổn hao (Lossless):} Mạng không tiêu hao năng lượng tuân theo điều kiện trực giao unita:
    \[
    [S]^H [S] = [I] \iff \sum_{k=1}^N S_{ki}^* S_{kj} = \delta_{ij}
    \]
    Nếu $[S]^H [S] \neq [I]$, mạng là mạng có tổn hao (lossy network).
    \item \textbf{Hệ số phản xạ ngõ vào khi tải không phối hợp:} Đối với mạng hai cửa có tải $\Gamma_L$ tại cửa 2:
    \[
    \Gamma_{in} = S_{11} + \frac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L}
    \]
    Và đối với ngõ ra khi nguồn có hệ số phản xạ $\Gamma_S$:
    \[
    \Gamma_{out} = S_{22} + \frac{S_{12}S_{21}\Gamma_S}{1 - S_{11}\Gamma_S}
    \]
\end{enumerate}

\section{Bài 1: Bộ suy giảm hình T (T-Attenuator)}

\subsection{Đề bài}
Một bộ suy giảm hình T gồm hai điện trở nối tiếp $R_1$ và một điện trở song song $R_2$ được thiết kế để phối hợp với $50~\Omega$ ở cả hai cửa và có độ suy giảm 6~dB (Hình~\ref{fig:bai1_schematic}).
\begin{enumerate}
    \item[(a)] Tính $R_1$ và $R_2$.
    \item[(b)] Viết ma trận $[S]$ của bộ suy giảm; mạng có thuận nghịch, phối hợp, không tổn hao không?
    \item[(c)] Cửa 2 được nối với tải $Z_L = 100~\Omega$. Tính hệ số phản xạ $\Gamma_{in}$ nhìn vào cửa 1 và suy hao phản hồi (return loss) tại cửa 1. So sánh với suy hao phản hồi khi nối tải trực tiếp vào nguồn và rút ra công dụng của bộ suy giảm.

\end{enumerate}

\subsection{Lời giải}

\subsubsection{(a) Xác định các giá trị điện trở $R_1$ và $R_2$}
Bộ suy giảm hình T đối xứng được thiết kế để phối hợp trở kháng chuẩn $Z_0 = 50~\Omega$ ở cả hai cửa. Độ suy giảm theo đơn vị decibel là $L = 6~\text{dB}$.
Tỉ số suy giảm điện áp $K$ được xác định theo:
\[
K = 10^{L/20} = 10^{6/20} = 10^{0{,}3} \approx 1{,}99526 \approx 2
\]
Công thức thiết kế kinh điển cho bộ suy giảm đối xứng hình T:
\[
R_1 = Z_0 \left( \frac{K - 1}{K + 1} \right), \qquad R_2 = Z_0 \left( \frac{2K}{K^2 - 1} \right)
\]
Thay giá trị chính xác $K = 10^{0{,}3} = 1{,}99526$ và $Z_0 = 50~\Omega$:
\[
R_1 = 50 \times \frac{1{,}99526 - 1}{1{,}99526 + 1} = 50 \times \frac{0{,}99526}{2{,}99526} \approx \mathbf{16{,}61~\Omega}
\]
\[
R_2 = 50 \times \frac{2 \times 1{,}99526}{1{,}99526^2 - 1} = 50 \times \frac{3{,}99052}{2{,}98107} \approx \mathbf{66{,}93~\Omega}
\]
\textit{(Nếu xấp xỉ lý thuyết $K \approx 2$: $R_1 = 50/3 \approx 16{,}67~\Omega$ và $R_2 = 50 \times (4/3) \approx 66{,}67~\Omega$)}.

\paragraph{Kiểm tra lại tính phối hợp:} Khi cửa 2 nối tải $Z_0 = 50~\Omega$, trở kháng nhìn vào cửa 1 là:
\[
Z_{in} = R_1 + \left[ R_2 \parallel (R_1 + Z_0) \right] = 16{,}61 + \frac{66{,}93 \times (16{,}61 + 50)}{66{,}93 + 16{,}61 + 50} = 16{,}61 + 33{,}39 = 50{,}00~\Omega \quad (\checkmark)
\]
Vậy:
\[
\boxed{R_1 = 16{,}61~\Omega, \quad R_2 = 66{,}93~\Omega}
\]

\subsubsection{(b) Ma trận tán xạ $[S]$ và các đặc tính của mạng}
\begin{itemize}
    \item Do mạng đối xứng và phối hợp hoàn hảo với $Z_0 = 50~\Omega$ ở cả hai cửa khi cửa kia kết cuối chuẩn:
    \[
    S_{11} = S_{22} = 0
    \]
    \item Độ suy giảm $6~\text{dB}$ tương ứng với biên độ hệ số truyền đạt:
    \[
    |S_{21}| = |S_{12}| = 10^{-L/20} = 10^{-6/20} \approx 0{,}5012 \approx 0{,}50
    \]
    Do mạng thuần trở (không gây lệch pha tín hiệu), các tham số là số thực dương:
    \[
    S_{21} = S_{12} = 0{,}5012
    \]
\end{itemize}
Ma trận tán xạ của bộ suy giảm:
\[
\boxed{[S] = \begin{bmatrix} 0 & 0{,}5012 \\ 0{,}5012 & 0 \end{bmatrix}}
\]

\paragraph{Đánh giá các đặc tính mạng:}
\begin{enumerate}
    \item \textbf{Tính thuận nghịch (Reciprocal):} Ma trận $[S]$ đối xứng qua đường chéo chính ($S_{12} = S_{21} = 0{,}5012$). Mạng chỉ gồm các phần tử điện trở tuyến tính đẳng hướng thụ động. Kết luận: \textbf{Mạng thuận nghịch}.
    \item \textbf{Tính phối hợp (Matched):} $S_{11} = S_{22} = 0$. Không có phản xạ khi các cửa được kết cuối $50~\Omega$. Kết luận: \textbf{Mạng phối hợp hoàn hảo ở cả hai cửa}.
    \item \textbf{Tính không tổn hao (Lossless):} Kiểm tra điều kiện Unita $[S]^H [S] = [I]$:
    \[
    [S]^H [S] = \begin{bmatrix} 0 & 0{,}5012 \\ 0{,}5012 & 0 \end{bmatrix} \begin{bmatrix} 0 & 0{,}5012 \\ 0{,}5012 & 0 \end{bmatrix} = \begin{bmatrix} 0{,}2512 & 0 \\ 0 & 0{,}2512 \end{bmatrix} \neq \begin{bmatrix} 1 & 0 \\ 0 & 1 \end{bmatrix}
    \]
    Do $[S]^H [S] \neq [I]$ và tổng công suất truyền qua nhỏ hơn công suất tới ($|S_{21}|^2 \approx 0{,}2512 < 1$), kết luận: \textbf{Mạng có tổn hao} (năng lượng tiêu tán thành nhiệt trên $R_1, R_2$).
\end{enumerate}

\subsubsection{(c) Tải $Z_L = 100~\Omega$: Tính $\Gamma_{in}$, $RL$ và công dụng của bộ suy giảm}
Hệ số phản xạ tại tải $Z_L = 100~\Omega$ đối với trở kháng đặc tính chuẩn $Z_0 = 50~\Omega$:
\[
\Gamma_L = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{100 - 50}{100 + 50} = \frac{50}{150} = \frac{1}{3} \approx 0{,}3333
\]
Hệ số phản xạ nhìn vào cửa 1 của mạng hai cửa:
\[
\Gamma_{in} = S_{11} + \frac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L}
\]
Thay các giá trị $S_{11} = S_{22} = 0$ và $S_{12} = S_{21} = 0{,}5012$:
\[
\Gamma_{in} = 0 + \frac{(0{,}5012)^2 \times (1/3)}{1 - 0 \times (1/3)} = 0{,}2512 \times \frac{1}{3} = \mathbf{0{,}0837}
\]
Suy hao phản hồi (Return Loss) tại cửa 1:
\[
RL_1 = -20 \log_{10} |\Gamma_{in}| = -20 \log_{10}(0{,}0837) = \mathbf{21{,}55~\text{dB}}
\]

\paragraph{So sánh khi nối tải trực tiếp vào nguồn (không có bộ suy giảm):}
\[
\Gamma_{\text{direct}} = \Gamma_L = \frac{1}{3} \approx 0{,}3333
\]
\[
RL_{\text{direct}} = -20 \log_{10} |\Gamma_L| = -20 \log_{10}(0{,}3333) = \mathbf{9{,}54~\text{dB}}
\]

\paragraph{Nhận xét và rút ra công dụng của bộ suy giảm:}
\begin{itemize}
    \item Suy hao phản hồi được cải thiện vượt bậc: tăng từ $9{,}54~\text{dB}$ lên $21{,}55~\text{dB}$, tức cải thiện đúng:
    \[
    \Delta RL = 21{,}55 - 9{,}54 = 12{,}01~\text{dB} \approx 2 \times L = 2 \times 6~\text{dB}
    \]
    \item \textbf{Cơ chế vật lý và công dụng:} Khi tải bị lệch trở kháng, sóng phản xạ từ tải phải đi qua bộ suy giảm hai lần (lần đi bị suy giảm $6~\text{dB}$, phản xạ tại tải, và lần về bị suy giảm thêm $6~\text{dB}$ nữa). Do đó, biên độ sóng phản xạ quay về nguồn giảm $12~\text{dB}$.
    \item \textbf{Ứng dụng thực tế:} Bộ suy giảm được sử dụng rộng rãi như một \textit{tầng đệm phối hợp trở kháng} (matching pad) giữa nguồn phát (hoặc bộ khuếch đại nhạy cảm) và tải không phối hợp để bảo vệ nguồn, chống sóng đứng và triệt tiêu phản xạ.
\end{itemize}
\[
\boxed{\Gamma_{in} = 0{,}0837, \quad RL_1 = 21{,}55~\text{dB} \quad (\text{Cải thiện } 12{,}01~\text{dB} \text{ so với trực tiếp})}
\]

\subsubsection{(d) Phân tích cán cân công suất với công suất tới $P_{inc} = 1~\text{mW}$}
Cho công suất sóng tới tại cửa 1 là $P_{inc} = 1~\text{mW} = 1000~\mu\text{W}$.
\begin{enumerate}
    \item \textbf{Công suất phản xạ tại cửa 1 ($P_{ref}$):}
    \[
    P_{ref} = |\Gamma_{in}|^2 P_{inc} = (0{,}08373)^2 \times 1~\text{mW} = 0{,}00701~\text{mW} \approx \mathbf{7{,}01~\mu\text{W}}
    \]
    \item \textbf{Công suất đi vào bộ suy giảm ($P_{in}$):}
    \[
    P_{in} = P_{inc} (1 - |\Gamma_{in}|^2) = 1 - 0{,}00701 = 0{,}99299~\text{mW} \approx \mathbf{993{,}0~\mu\text{W}}
    \]
    \item \textbf{Công suất hấp thụ bởi tải $Z_L = 100~\Omega$ ($P_L$):}
    Độ lợi chuyển đổi (Transducer Gain) từ nguồn $50~\Omega$ ($\Gamma_S = 0$) đến tải $\Gamma_L = 1/3$:
    \[
    G_T = \frac{|S_{21}|^2 (1 - |\Gamma_S|^2)(1 - |\Gamma_L|^2)}{|1 - S_{22}\Gamma_L|^2 |1 - \Gamma_{in}\Gamma_S|^2} = |S_{21}|^2 (1 - |\Gamma_L|^2)
    \]
    \[
    G_T = (0{,}5012)^2 \times \left(1 - \frac{1}{9}\right) = 0{,}2512 \times \frac{8}{9} = 0{,}22329
    \]
    Do đó công suất đến tải là:
    \[
    P_L = G_T \times P_{inc} = 0{,}22329 \times 1~\text{mW} = 0{,}2233~\text{mW} \approx \mathbf{223{,}3~\mu\text{W}}
    \]
    \item \textbf{Công suất tiêu tán trong bộ suy giảm ($P_{diss}$):}
    Theo định luật bảo toàn năng lượng, công suất tiêu hao dưới dạng nhiệt trong các điện trở của bộ suy giảm là hiệu giữa công suất thực đi vào và công suất tải nhận được:
    \[
    P_{diss} = P_{in} - P_L = 0{,}9930~\text{mW} - 0{,}2233~\text{mW} = 0{,}7697~\text{mW} \approx \mathbf{769{,}7~\mu\text{W}}
    \]
\end{enumerate}

\paragraph{Kiểm tra cân bằng năng lượng toàn hệ thống:}
\[
P_{ref} + P_L + P_{diss} = 0{,}00701 + 0{,}22329 + 0{,}76970 = 1{,}0000~\text{mW} = P_{inc} \quad (\checkmark)
\]
\[
\boxed{P_{ref} = 7{,}01~\mu\text{W}, \quad P_L = 223{,}3~\mu\text{W}, \quad P_{diss} = 769{,}7~\mu\text{W}}
\]

\subsection{Lời giải}

\subsubsection{(a) Xác định trở kháng, chiều dài nhánh và điện trở cách ly}
\begin{enumerate}
    \item \textbf{Trở kháng đặc tính của hai nhánh biến đổi $\lambda/4$:}
    Theo cấu hình bộ chia Wilkinson chia đều công suất ($3~\text{dB}$):
    \[
    Z_{\text{branch}} = Z_0 \sqrt{2} = 50 \sqrt{2} \approx \mathbf{70{,}71~\Omega}
    \]
    \item \textbf{Chiều dài vật lý của mỗi nhánh ($\ell = \lambda_g/4$):}
    Vận tốc ánh sáng trong chân không $c = 3 \times 10^8~\text{m/s}$. Bước sóng trong đường truyền vi dải với hằng số điện môi hiệu dụng $\varepsilon_{eff} = 3{,}17$ ở tần số $f = 2~\text{GHz}$:
    \[
    \lambda_g = \frac{c}{f\sqrt{\varepsilon_{eff}}} = \frac{3 \times 10^8}{2 \times 10^9 \times \sqrt{3{,}17}} = \frac{0{,}15}{1{,}78045} \approx 0{,}08425~\text{m} = 84{,}25~\text{mm}
    \]
    Chiều dài vật lý mỗi nhánh một phần tư bước sóng:
    \[
    \ell = \frac{\lambda_g}{4} = \frac{84{,}25~\text{mm}}{4} \approx \mathbf{21{,}06~\text{mm}}
    \]
    \item \textbf{Giá trị điện trở cách ly $R$:}
    Để triệt tiêu phản xạ và cách ly giữa hai cửa ngõ ra (ở mode lẻ):
    \[
    R = 2 Z_0 = 2 \times 50 = \mathbf{100~\Omega}
    \]
\end{enumerate}
\[
\boxed{Z_{\text{branch}} = 70{,}71~\Omega, \quad \ell = 21{,}06~\text{mm}, \quad R = 100~\Omega}
\]

\subsubsection{(b) Phân tích ý nghĩa vật lý của ma trận tán xạ $[S]$}
Ma trận tán xạ tại tần số thiết kế:
\[
[S] = \frac{-j}{\sqrt{2}} \begin{bmatrix} 0 & 1 & 1 \\ 1 & 0 & 0 \\ 1 & 0 & 0 \end{bmatrix} = \begin{bmatrix} 0 & -j/\sqrt{2} & -j/\sqrt{2} \\ -j/\sqrt{2} & 0 & 0 \\ -j/\sqrt{2} & 0 & 0 \end{bmatrix}
\]
\paragraph{Ý nghĩa vật lý của từng phần tử:}
\begin{itemize}
    \item $S_{11} = 0$: Ngõ vào (cửa 1) phối hợp trở kháng hoàn hảo với $50~\Omega$ khi hai cửa ra kết cuối $50~\Omega$.
    \item $S_{22} = S_{33} = 0$: Hai ngõ ra (cửa 2 và cửa 3) cũng phối hợp trở kháng hoàn hảo với $50~\Omega$ khi các cửa còn lại kết cuối chuẩn.
    \item $S_{21} = S_{31} = \frac{-j}{\sqrt{2}} = \frac{1}{\sqrt{2}} \angle -90^\circ$: Tín hiệu cấp vào cửa 1 được chia công suất đồng đều cho cửa 2 và cửa 3. Biên độ công suất mỗi cửa là $|S_{21}|^2 = 1/2 = -3~\text{dB}$. Thừa số $-j = e^{-j90^\circ}$ biểu thị độ lệch pha trễ $90^\circ$ do sóng truyền qua đoạn đường truyền dài $\lambda_g/4$.
    \item $S_{12} = S_{13} = \frac{-j}{\sqrt{2}}$: Hệ số truyền theo chiều ngược lại bằng chiều thuận do mạng có tính thuận nghịch.
    \item $S_{23} = S_{32} = 0$: \textbf{Độ cách ly lý tưởng} giữa cửa 2 và cửa 3. Sóng cấp vào cửa 2 không bị rò sang cửa 3 và ngược lại, nhờ sự giao thoa triệt tiêu giữa sóng truyền qua hai đoạn $\lambda/4$ và sóng đi qua điện trở cách ly $R$.
\end{itemize}

\paragraph{Đặc tính thuận nghịch và tổn hao:}
\begin{itemize}
    \item \textbf{Thuận nghịch:} Vì $S_{ij} = S_{ji} \; \forall i, j$ ($[S] = [S]^T$), mạng là \textbf{thuận nghịch}.
    \item \textbf{Tổn hao:} Xét tích $[S]^H [S]$:
    \[
    [S]^H [S] = \left(\frac{1}{\sqrt{2}}\right)^2 \begin{bmatrix} 0 & 1 & 1 \\ 1 & 0 & 0 \\ 1 & 0 & 0 \end{bmatrix} \begin{bmatrix} 0 & 1 & 1 \\ 1 & 0 & 0 \\ 1 & 0 & 0 \end{bmatrix} = \frac{1}{2} \begin{bmatrix} 2 & 0 & 0 \\ 0 & 1 & 1 \\ 0 & 1 & 1 \end{bmatrix} = \begin{bmatrix} 1 & 0 & 0 \\ 0 & 0{,}5 & 0{,}5 \\ 0 & 0{,}5 & 0{,}5 \end{bmatrix} \neq [I]
    \]
    Kết luận: \textbf{Mạng có tổn hao} do sự hiện diện của điện trở $R$. Tuy nhiên, khi kích thích đối xứng tại cửa 1 (mode chẵn hoàn toàn), không có dòng điện chạy qua $R$, nên bộ chia không tiêu hao năng lượng trong chế độ hoạt động bình thường ($|S_{21}|^2 + |S_{31}|^2 = 0{,}5 + 0{,}5 = 1$). Tổn hao chỉ xảy ra khi kích thích từ cửa ra có chứa thành phần mode lẻ.
\end{itemize}

\subsubsection{(c) Tính công suất ra và công suất tiêu tán trên $R$ trong 4 trường hợp}
Mối quan hệ sóng: $\mathbf{b} = [S]\mathbf{a}$.
\begin{enumerate}
    \item \textbf{Trường hợp 1: Cấp $1~\text{W}$ vào cửa 1 ($a_1 = 1$, $a_2 = a_3 = 0$):}
    \[
    \begin{bmatrix} b_1 \\ b_2 \\ b_3 \end{bmatrix} = [S] \begin{bmatrix} 1 \\ 0 \\ 0 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 0 \\ 1 \\ 1 \end{bmatrix}
    \]
    \begin{itemize}
        \item $P_1 = |b_1|^2 = 0~\text{W}$ (không phản xạ).
        \item $P_2 = |b_2|^2 = |-j/\sqrt{2}|^2 = \mathbf{0{,}5~\text{W}}$.
        \item $P_3 = |b_3|^2 = |-j/\sqrt{2}|^2 = \mathbf{0{,}5~\text{W}}$.
        \item Công suất tiêu tán trên $R$: $P_R = P_{in} - (P_1 + P_2 + P_3) = 1 - (0 + 0{,}5 + 0{,}5) = \mathbf{0~\text{W}}$.
    \end{itemize}

    \item \textbf{Trường hợp 2: Cấp $1~\text{W}$ vào cửa 2 ($a_1 = 0$, $a_2 = 1$, $a_3 = 0$):}
    \[
    \begin{bmatrix} b_1 \\ b_2 \\ b_3 \end{bmatrix} = [S] \begin{bmatrix} 0 \\ 1 \\ 0 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 1 \\ 0 \\ 0 \end{bmatrix}
    \]
    \begin{itemize}
        \item $P_1 = |b_1|^2 = |-j/\sqrt{2}|^2 = \mathbf{0{,}5~\text{W}}$.
        \item $P_2 = |b_2|^2 = 0~\text{W}$ (cửa 2 phối hợp).
        \item $P_3 = |b_3|^2 = 0~\text{W}$ (cửa 3 cách ly hoàn toàn).
        \item Công suất tiêu tán trên $R$: $P_R = 1 - 0{,}5 = \mathbf{0{,}5~\text{W}}$ (phân nửa công suất bị triệt tiêu trên $R$).
    \end{itemize}

    \item \textbf{Trường hợp 3: Cấp $1~\text{W}$ vào cả hai cửa 2 và 3 cùng pha ($a_2 = 1, a_3 = 1, a_1 = 0$):}
    \[
    \begin{bmatrix} b_1 \\ b_2 \\ b_3 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 0 & 1 & 1 \\ 1 & 0 & 0 \\ 1 & 0 & 0 \end{bmatrix} \begin{bmatrix} 0 \\ 1 \\ 1 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 2 \\ 0 \\ 0 \end{bmatrix} = \begin{bmatrix} -j\sqrt{2} \\ 0 \\ 0 \end{bmatrix}
    \]
    \begin{itemize}
        \item $P_1 = |b_1|^2 = |-j\sqrt{2}|^2 = \mathbf{2~\text{W}}$.
        \item $P_2 = |b_2|^2 = 0~\text{W}, \quad P_3 = |b_3|^2 = 0~\text{W}$.
        \item Tổng công suất vào là $P_{in} = |a_2|^2 + |a_3|^2 = 1 + 1 = 2~\text{W}$.
        \item Công suất tiêu tán trên $R$: $P_R = 2 - 2 = \mathbf{0~\text{W}}$ (bộ cộng công suất không tổn hao ở mode chẵn).
    \end{itemize}

    \item \textbf{Trường hợp 4: Cấp $1~\text{W}$ vào mỗi cửa 2 và 3 ngược pha ($a_2 = 1, a_3 = -1, a_1 = 0$):}
    \[
    \begin{bmatrix} b_1 \\ b_2 \\ b_3 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 0 & 1 & 1 \\ 1 & 0 & 0 \\ 1 & 0 & 0 \end{bmatrix} \begin{bmatrix} 0 \\ 1 \\ -1 \end{bmatrix} = \frac{-j}{\sqrt{2}} \begin{bmatrix} 0 \\ 0 \\ 0 \end{bmatrix} = \begin{bmatrix} 0 \\ 0 \\ 0 \end{bmatrix}
    \]
    \begin{itemize}
        \item $P_1 = P_2 = P_3 = \mathbf{0~\text{W}}$.
        \item Tổng công suất vào: $P_{in} = 1 + 1 = 2~\text{W}$.
        \item Công suất tiêu tán trên $R$: $P_R = 2 - 0 = \mathbf{2~\text{W}}$ (ở mode lẻ, toàn bộ công suất triệt tiêu nhau và hấp thụ hoàn toàn trên điện trở $R$).
    \end{itemize}
\end{enumerate}

\subsubsection{(d) Phân tích khi cửa 3 bị nối nhầm với tải $Z_L = 75~\Omega$}
Hệ số phản xạ tại cửa 3:
\[
\Gamma_3 = \frac{75 - 50}{75 + 50} = \frac{25}{125} = 0{,}2
\]
Cửa 2 nối tải phối hợp $50~\Omega \implies \Gamma_2 = 0$.
\paragraph{1. Tính hệ số phản xạ nhìn vào cửa 1 ($\Gamma_{in}$):}
Tín hiệu tới cửa 1 ($a_1$) truyền sang cửa 3 tạo sóng ra $b_3 = S_{31} a_1$. Sóng này gặp tải $\Gamma_3$ bị phản xạ trở lại cửa 3 thành sóng tới $a_3 = \Gamma_3 b_3 = S_{31} \Gamma_3 a_1$. Sóng tới cửa 3 này phản xạ quay lại cửa 1 qua hệ số $S_{13}$:
\[
\Gamma_{in} = \frac{b_1}{a_1} = S_{11} + \frac{S_{13} \Gamma_3 S_{31}}{1 - S_{33}\Gamma_3}
\]
Thay số với $S_{11} = S_{33} = 0$, $S_{31} = S_{13} = -j/\sqrt{2}$:
\[
\Gamma_{in} = 0 + \frac{\left( \frac{-j}{\sqrt{2}} \right)^2 \times 0{,}2}{1 - 0} = \left(-\frac{1}{2}\right) \times 0{,}2 = \mathbf{-0{,}1}
\]
Suy hao phản hồi tại cửa 1: $RL_1 = -20 \log_{10}(|-0{,}1|) = 20~\text{dB}$.

\paragraph{2. Tính hệ số truyền đến cửa 2 ($S_{21,\text{eff}}$):}
Hệ số truyền từ cửa 1 đến cửa 2 khi cửa 3 bị phản xạ:
\[
S_{21,\text{eff}} = \frac{b_2}{a_1} = S_{21} + \frac{S_{23} \Gamma_3 S_{31}}{1 - S_{33}\Gamma_3}
\]
Nhờ đặc tính cách ly lý tưởng của bộ chia Wilkinson, $S_{23} = 0$, do đó:
\[
S_{21,\text{eff}} = S_{21} + 0 = S_{21} = \frac{-j}{\sqrt{2}} \approx \mathbf{0{,}7071 \angle -90^\circ}
\]
\textbf{Nhận xét ý nghĩa kỹ thuật:} Tải nối nhầm ở cửa 3 làm suy giảm độ phối hợp tại ngõ vào ($\Gamma_{in} = -0{,}1 \neq 0$), nhưng \textbf{hoàn toàn không làm ảnh hưởng đến hệ số truyền và công suất ra ở cửa 2} ($|S_{21,\text{eff}}|^2 = 0{,}5 = -3~\text{dB}$). Đây chính là ưu thế xuất sắc của bộ chia Wilkinson so với bộ chia điện trở.
\[
\boxed{\Gamma_{in} = -0{,}1, \quad S_{21,\text{eff}} = \frac{-j}{\sqrt{2}} \approx 0{,}7071\angle -90^\circ}
\]

    Z_{\text{branch}, 2} = R + Z_0 = \frac{Z_0}{3} + Z_0 = \frac{4}{3} Z_0 = 16{,}67 + 50 = 66{,}67~\Omega
    \]
    \item Trở kháng nhìn từ nút trung tâm về phía cửa 3:
    \[
    Z_{\text{branch}, 3} = R + Z_0 = \frac{4}{3} Z_0 = 66{,}67~\Omega
    \]
    \item Hai nhánh này mắc song song với nhau tại nút trung tâm:
    \[
    Z_{\text{node}} = Z_{\text{branch}, 2} \parallel Z_{\text{branch}, 3} = \frac{\frac{4}{3}Z_0}{2} = \frac{2}{3} Z_0 = 33{,}33~\Omega
    \]
    \item Trở kháng ngõ vào tổng cộng nhìn từ cửa 1:
    \[
    Z_{in,1} = R + Z_{\text{node}} = \frac{Z_0}{3} + \frac{2}{3}Z_0 = Z_0 = \mathbf{50~\Omega}
    \]
\end{itemize}
Do $Z_{in,1} = Z_0$, cửa 1 hoàn toàn phối hợp ($S_{11} = 0$).
Vì mạch có cấu trúc đối xứng hoàn toàn đối với cả ba cửa, lập luận tương tự cho thấy:
\[
Z_{in,2} = Z_0 = 50~\Omega \implies S_{22} = 0, \qquad Z_{in,3} = Z_0 = 50~\Omega \implies S_{33} = 0
\]
Vậy mỗi cửa đều phối hợp hoàn hảo khi hai cửa còn lại kết cuối $50~\Omega$. $(\square)$

\subsubsection{(b) Xác định ma trận tán xạ $[S]$}
Từ kết quả phần (a): $S_{11} = S_{22} = S_{33} = 0$.
Xét hệ số truyền từ cửa 1 đến cửa 2 ($S_{21}$):
Đặt sóng tới $a_1$ vào cửa 1, các cửa 2 và 3 nối $Z_0$ ($a_2 = a_3 = 0$).
Vì cửa 1 phối hợp, điện áp tại cửa 1 là $V_1 = a_1 \sqrt{Z_0}$.
Điện áp tại nút trung tâm do phân áp giữa $R$ và $Z_{\text{node}}$:
\[
V_{\text{node}} = V_1 \times \frac{Z_{\text{node}}}{R + Z_{\text{node}}} = V_1 \times \frac{\frac{2}{3}Z_0}{\frac{1}{3}Z_0 + \frac{2}{3}Z_0} = \frac{2}{3} V_1
\]
Điện áp tại cửa 2 do phân áp giữa $R$ và $Z_0$:
\[
V_2 = V_{\text{node}} \times \frac{Z_0}{R + Z_0} = \left(\frac{2}{3} V_1\right) \times \frac{Z_0}{\frac{4}{3}Z_0} = \frac{2}{3} \times \frac{3}{4} \times V_1 = \frac{1}{2} V_1
\]
Sóng phản xạ ra tại cửa 2:
\[
b_2 = \frac{V_2}{\sqrt{Z_0}} = \frac{1}{2} \frac{V_1}{\sqrt{Z_0}} = \frac{1}{2} a_1 \implies S_{21} = \frac{b_2}{a_1} = \frac{1}{2}
\]
Do mạch có tính đối xứng vòng:
\[
S_{31} = S_{12} = S_{32} = S_{13} = S_{23} = \frac{1}{2}
\]
Ma trận tán xạ đầy đủ của bộ chia điện trở 3 cửa:
\[
\boxed{[S] = \begin{bmatrix} 0 & 1/2 & 1/2 \\ 1/2 & 0 & 1/2 \\ 1/2 & 1/2 & 0 \end{bmatrix}}
\]

\subsubsection{(c) Công suất ra, công suất tiêu tán và độ cách ly}
Khi cấp công suất $P_{in}$ vào cửa 1:
\begin{enumerate}
    \item \textbf{Công suất ra ở cửa 2 và cửa 3:}
    \[
    P_2 = |S_{21}|^2 P_{in} = \left(\frac{1}{2}\right)^2 P_{in} = \frac{1}{4} P_{in} \implies 10\log_{10}(0{,}25) = \mathbf{-6{,}02~\text{dB}}
    \]
    \[
    P_3 = |S_{31}|^2 P_{in} = \left(\frac{1}{2}\right)^2 P_{in} = \frac{1}{4} P_{in} \implies 10\log_{10}(0{,}25) = \mathbf{-6{,}02~\text{dB}}
    \]
    \item \textbf{Công suất bị tiêu tán trong mạch ($P_{diss}$):}
    Tổng công suất ngõ ra: $P_{\text{out}} = P_2 + P_3 = \frac{1}{4} P_{in} + \frac{1}{4} P_{in} = \frac{1}{2} P_{in}$.
    Công suất tiêu tán nhiệt trên 3 điện trở:
    \[
    P_{diss} = P_{in} - P_{\text{out}} = P_{in} - \frac{1}{2}P_{in} = \frac{1}{2} P_{in} = \mathbf{50\%}
    \]
    \item \textbf{Độ cách ly (Isolation) giữa cửa 2 và cửa 3:}
    \[
    \text{Isolation} = -20 \log_{10}|S_{23}| = -20 \log_{10}\left(\frac{1}{2}\right) = \mathbf{6{,}02~\text{dB}}
    \]
\end{enumerate}
\textit{Nhận xét:} Bộ chia điện trở chỉ có độ cách ly $6~\text{dB}$ (rất thấp), đồng thời tổn hao đến một nửa công suất vào các điện trở.

\subsubsection{(d) Ma trận tán xạ rút gọn khi cửa 3 nối tải $75~\Omega$ và so sánh với Bài 2(d)}
Hệ số phản xạ tại cửa 3 với tải $Z_L = 75~\Omega$:
\[
\Gamma_3 = \frac{75 - 50}{75 + 50} = \frac{25}{125} = 0{,}2
\]
Áp dụng công thức rút gọn ma trận 3 cửa thành 2 cửa khi cửa 3 nối tải $\Gamma_3$:

S'_{ij} = S_{ij} + \frac{S_{i3} \Gamma_3 S_{3j}}{1 - S_{33}\Gamma_3}
\]
Do $S_{33} = 0$, mẫu số luôn bằng $1$. Ta có:
\begin{align*}
S'_{11} &= S_{11} + S_{13}\Gamma_3 S_{31} = 0 + \left(\frac{1}{2}\right)(0{,}2)\left(\frac{1}{2}\right) = \mathbf{0{,}05} \\
S'_{22} &= S_{22} + S_{23}\Gamma_3 S_{32} = 0 + \left(\frac{1}{2}\right)(0{,}2)\left(\frac{1}{2}\right) = \mathbf{0{,}05} \\
S'_{21} &= S_{21} + S_{23}\Gamma_3 S_{31} = \frac{1}{2} + \left(\frac{1}{2}\right)(0{,}2)\left(\frac{1}{2}\right) = 0{,}5 + 0{,}05 = \mathbf{0{,}55} \\
S'_{12} &= S'_{21} = \mathbf{0{,}55}
\end{align*}
Ma trận tán xạ rút gọn hai cửa (cửa 1 -- cửa 2):
\[
\boxed{[S]' = \begin{bmatrix} 0{,}05 & 0{,}55 \\ 0{,}55 & 0{,}05 \end{bmatrix}}
\]

\subsection{Lời giải}

\subsubsection{(a) Trường hợp $Z_L = 50~\Omega$: Tính $\Gamma_{in}$}
Khi tải phối hợp với trở kháng chuẩn $Z_0 = 50~\Omega$:
\[
\Gamma_L = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{50 - 50}{50 + 50} = 0
\]
Hệ số phản xạ ngõ vào tổng quát của mạng hai cửa:
\[
\Gamma_{in} = S_{11} + \frac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L}
\]
Vì $\Gamma_L = 0$, số hạng thứ hai triệt tiêu hoàn toàn:
\[
\boxed{\Gamma_{in} = S_{11} = 0{,}6 \angle -160^\circ = -0{,}5638 - j0{,}2052}
\]
Trở kháng ngõ vào tương ứng:
\[
Z_{in} = Z_0 \frac{1 + \Gamma_{in}}{1 - \Gamma_{in}} = 50 \times \frac{1 + 0{,}6\angle -160^\circ}{1 - 0{,}6\angle -160^\circ} = 50 \times \frac{0{,}4362 - j0{,}2052}{1{,}5638 + j0{,}2052} \approx \mathbf{12{,}86 - j8{,}25~\Omega}
\]

\subsubsection{(b) Trường hợp $Z_L = 25~\Omega$: Tính $\Gamma_L$ và $\Gamma_{in}$}
\paragraph{1. Hệ số phản xạ của tải $Z_L = 25~\Omega$:}
\[
\boxed{\Gamma_L = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{25 - 50}{25 + 50} = -\frac{25}{75} = -\frac{1}{3} \approx -0{,}3333 = 0{,}3333\angle 180^\circ}
\]

\paragraph{2. Tính hệ số phản xạ ngõ vào $\Gamma_{in}$:}
Áp dụng công thức biến đổi trở kháng ngõ vào mạng hai cửa:
\[
\Gamma_{in} = S_{11} + \frac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L}
\]
Ta tính chi tiết từng đại lượng phức:
\begin{itemize}
    \item Tích $S_{12}S_{21}$:
    \[
    S_{12}S_{21} = (0{,}05\angle 40^\circ)(4\angle 60^\circ) = 0{,}2 \angle 100^\circ = 0{,}2(\cos 100^\circ + j\sin 100^\circ) = -0{,}03473 + j0{,}19696
    \]
    \item Tử số $S_{12}S_{21}\Gamma_L$:
    \[
    S_{12}S_{21}\Gamma_L = (0{,}2\angle 100^\circ) \left(-\frac{1}{3}\right) = \frac{0{,}2}{3} \angle (100^\circ - 180^\circ) = 0{,}06667 \angle -80^\circ
    \]
    \[
    = 0{,}06667(\cos(-80^\circ) + j\sin(-80^\circ)) = 0{,}01158 - j0{,}06565
    \]
    \item Đại lượng $S_{22}\Gamma_L$:
    \[
    S_{22}\Gamma_L = (0{,}4\angle -45^\circ) \left(-\frac{1}{3}\right) = \frac{0{,}4}{3} \angle (-45^\circ + 180^\circ) = 0{,}13333 \angle 135^\circ
    \]
    \[
    = 0{,}13333\left(-\frac{\sqrt{2}}{2} + j\frac{\sqrt{2}}{2}\right) = -0{,}09428 + j0{,}09428
    \]
    \item Mẫu số $1 - S_{22}\Gamma_L$:
    \[
    1 - S_{22}\Gamma_L = 1 - (-0{,}09428 + j0{,}09428) = 1{,}09428 - j0{,}09428
    \]
    Mô-đun bình phương của mẫu số:
    \[
    |1 - S_{22}\Gamma_L|^2 = (1{,}09428)^2 + (-0{,}09428)^2 = 1{,}19745 + 0{,}00889 = 1{,}20634
    \]
    \item Tỉ số hiệu chỉnh tải:
    \[
    \Delta \Gamma = \frac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L} = \frac{(0{,}01158 - j0{,}06565)(1{,}09428 + j0{,}09428)}{1{,}20634}
    \]
    Tử số:
    \[
    (0{,}01158 \times 1{,}09428 + 0{,}06565 \times 0{,}09428) + j(0{,}01158 \times 0{,}09428 - 0{,}06565 \times 1{,}09428)
    \]
    \[
    = (0{,}01267 + 0{,}00619) + j(0{,}00109 - 0{,}07184) = 0{,}01886 - j0{,}07075
    \]
    Chia cho mẫu số:
    \[
    \Delta \Gamma = \frac{0{,}01886 - j0{,}07075}{1{,}20634} = 0{,}01563 - j0{,}05865
    \]
    \item Cộng với $S_{11} = -0{,}56382 - j0{,}20521$:
    \[
    \Gamma_{in} = (-0{,}56382 + 0{,}01563) + j(-0{,}20521 - 0{,}05865) = \mathbf{-0{,}54819 - j0{,}26386}
    \]
    Biên độ và góc pha:
    \[
    |\Gamma_{in}| = \sqrt{(-0{,}54819)^2 + (-0{,}26386)^2} = \sqrt{0{,}30051 + 0{,}06962} = \sqrt{0{,}37013} \approx \mathbf{0{,}6084}
    \]
    \[
    \angle \Gamma_{in} = \arctan\left(\frac{-0{,}26386}{-0{,}54819}\right) - 180^\circ = 25{,}70^\circ - 180^\circ = \mathbf{-154{,}30^\circ}
    \]
\end{itemize}
\[
\boxed{\Gamma_L = -\frac{1}{3} \approx -0{,}3333, \quad \Gamma_{in} = 0{,}608\angle -154{,}3^\circ}
\]

\subsubsection{(c) Tính $\Gamma_{out}$ và nhận xét về việc phối hợp trở kháng}
Hệ số phản xạ nhìn vào ngõ ra transistor:
\[
\Gamma_{out} = S_{22} + \frac{S_{12}S_{21}\Gamma_S}{1 - S_{11}\Gamma_S}
\]
Do nguồn tín hiệu có nội trở $R_s = 50~\Omega = Z_0 \implies \Gamma_S = 0$, ta có:
\[
\boxed{\Gamma_{out} = S_{22} = 0{,}4 \angle -45^\circ = 0{,}2828 - j0{,}2828}
\]
Trở kháng ngõ ra của transistor:
\[
Z_{out} = Z_0 \frac{1 + \Gamma_{out}}{1 - \Gamma_{out}} = 50 \times \frac{1 + 0{,}4\angle -45^\circ}{1 - 0{,}4\angle -45^\circ} \approx \mathbf{70{,}66 - j47{,}58~\Omega}
\]

\paragraph{Nhận xét sâu sắc về thiết kế phối hợp trở kháng hai phía:}
\begin{enumerate}
    \item \textbf{Mức độ không phối hợp cao:} 
    \begin{itemize}
        \item Tại ngõ vào: $|\Gamma_{in}| \approx 0{,}608$ (tương ứng $RL \approx 4{,}3~\text{dB}$, VSWR $\approx 4{,}1$), tổn hao do phản xạ rất lớn ($|\Gamma|^2 \approx 37\%$). Cần có mạch phối hợp trở kháng ngõ vào (Input Matching Network - IMN) biến đổi nguồn $50~\Omega$ thành $\Gamma_S = \Gamma_{in}^*$.
        \item Tại ngõ ra: $|\Gamma_{out}| = 0{,}4$ (tương ứng VSWR $\approx 2{,}33$), cần có mạch phối hợp ngõ ra (Output Matching Network - OMN) biến đổi tải thành $\Gamma_L = \Gamma_{out}^*$.
    \end{itemize}
    \item \textbf{Hiệu ứng truyền ngược (Non-unilateral behavior):}
    Do $S_{12} = 0{,}05 \neq 0$, transistor không phải là linh kiện đơn hướng hoàn toàn. Thay đổi tải $\Gamma_L$ sẽ làm thay đổi trở kháng vào $\Gamma_{in}$ (như đã thấy ở câu b khi $\Gamma_{in}$ đổi từ $0{,}6\angle -160^\circ$ sang $0{,}608\angle -154{,}3^\circ$). Vì vậy, việc thiết kế IMN và OMN có sự tương tác qua lại, cần thiết kế phối hợp liên hợp đồng thời (simultaneous conjugate matching) hoặc dùng phương pháp lặp.
    \item \textbf{Tính ổn định (Stability):}
    Vì $|S_{11}| < 1$ và $|S_{22}| < 1$, nhưng do $S_{12} \neq 0$, cần kiểm tra hệ số ổn định Rollett ($K$) và $\Delta$ để đảm bảo transistor hoạt động trong vùng ổn định tuyệt đối trước khi tiến hành phối hợp trở kháng.
\end{enumerate}

\section{Bài 5: Ghép nối chuỗi các phần tử (Cascade)}

\subsection{Lời giải}

\subsubsection{(a) Xác định các hệ số phản xạ $\Gamma_C, \Gamma_B, \Gamma_A$}
Ta tiến hành tính toán lan truyền ngược từ tải $Z_L$ về phía nguồn:
\begin{enumerate}
    \item \textbf{Tại mặt phẳng C (ngay trước tải $Z_L = 25~\Omega$):}
    \[
    \boxed{\Gamma_C = \frac{Z_L - Z_0}{Z_L + Z_0} = \frac{25 - 50}{25 + 50} = -\frac{1}{3} \approx \mathbf{-0{,}3333}}
    \]
    \item \textbf{Tại mặt phẳng B (trước bộ suy giảm $6~\text{dB}$):}
    Bộ suy giảm ở Bài 1 có các tham số tán xạ: $S_{11,att} = S_{22,att} = 0$, $S_{21,att} = S_{12,att} = 0{,}5012$.
    Hệ số phản xạ nhìn vào cửa 1 của bộ suy giảm khi cửa 2 nối tải $\Gamma_C$:
    \[
    \Gamma_B = S_{11,att} + \frac{S_{12,att} S_{21,att} \Gamma_C}{1 - S_{22,att}\Gamma_C} = 0 + \frac{(0{,}5012)^2 \times (-1/3)}{1 - 0}
    \]
    \[
    \boxed{\Gamma_B = -0{,}08373}
    \]
    \item \textbf{Tại mặt phẳng A (sau bộ khuếch đại, trước đoạn đường truyền):}
    Đoạn đường truyền không tổn hao có trở kháng đặc tính $Z_0 = 50~\Omega$ và độ dài điện $\theta = 45^\circ = \pi/4$.
    Quy tắc biến đổi hệ số phản xạ qua đoạn đường truyền:
    \[
    \Gamma_A = \Gamma_B \, e^{-j 2\theta} = -0{,}08373 \, e^{-j 2(45^\circ)} = -0{,}08373 \, e^{-j 90^\circ}
    \]
    Do $e^{-j 90^\circ} = -j$, ta có:
    \[
    \Gamma_A = -0{,}08373 \times (-j) = j 0{,}08373 = \mathbf{0{,}08373 \angle 90^\circ}
    \]
    \[
    \boxed{\Gamma_A = 0{,}0837 \angle 90^\circ = j0{,}0837}
    \]
\end{enumerate}

\subsubsection{(b) Tính $\Gamma_{in}$ tại ngõ vào bộ khuếch đại (mặt phẳng P)}
Tải nhìn từ ngõ ra của transistor tại mặt phẳng A chính là $\Gamma_L' = \Gamma_A = j 0{,}08373$.
Các thông số S của transistor (từ Bài 4):
\[
S_{11} = 0{,}6\angle -160^\circ, \quad S_{21} = 4\angle 60^\circ, \quad S_{12} = 0{,}05\angle 40^\circ, \quad S_{22} = 0{,}4\angle -45^\circ
\]
Hệ số phản xạ ngõ vào:
\[
\Gamma_{in} = S_{11} + \frac{S_{12}S_{21}\Gamma_A}{1 - S_{22}\Gamma_A}
\]
Tính toán chi tiết:
\begin{itemize}
    \item Mẫu số $1 - S_{22}\Gamma_A$:
    \[
    S_{22}\Gamma_A = (0{,}4\angle -45^\circ)(0{,}08373\angle 90^\circ) = 0{,}03349 \angle 45^\circ
    \]
    \[
    = 0{,}03349\left(\frac{\sqrt{2}}{2} + j\frac{\sqrt{2}}{2}\right) = 0{,}02368 + j0{,}02368
    \]
    \[
    1 - S_{22}\Gamma_A = (1 - 0{,}02368) - j0{,}02368 = 0{,}97632 - j0{,}02368
    \]
    Mô-đun bình phương của mẫu số:
    \[
    |1 - S_{22}\Gamma_A|^2 = (0{,}97632)^2 + (0{,}02368)^2 = 0{,}95320 + 0{,}00056 = 0{,}95376
    \]
    \item Tử số $S_{12}S_{21}\Gamma_A$:
    \[
    S_{12}S_{21} = 0{,}2 \angle 100^\circ
    \]
    \[
    S_{12}S_{21}\Gamma_A = (0{,}2\angle 100^\circ)(0{,}08373\angle 90^\circ) = 0{,}01675 \angle 190^\circ
    \]
    \[
    = 0{,}01675(\cos 190^\circ + j\sin 190^\circ) = -0{,}01650 - j0{,}02908
    \]
    \item Thương số:
    \[
    \Delta \Gamma = \frac{-0{,}01650 - j0{,}002908}{0{,}97632 - j0{,}02368} = \frac{(-0{,}01650 - j0{,}002908)(0{,}97632 + j0{,}02368)}{0{,}95376}
    \]
    \[
    = \frac{(-0{,}01611 + 0{,}00007) + j(-0{,}00039 - 0{,}00284)}{0{,}95376} = -0{,}01682 - j0{,}0339
    \]
    \item Cộng với $S_{11} = -0{,}56382 - j0{,}20521$:
    \[
    \Gamma_{in} = (-0{,}56382 - 0{,}01682) + j(-0{,}20521 - 0{,}00339) = \mathbf{-0{,}58064 - j0{,}20860}
    \]
    Biên độ và góc pha:
    \[
    |\Gamma_{in}| = \sqrt{(-0{,}58064)^2 + (-0{,}20860)^2} = \sqrt{0{,}33714 + 0{,}04351} = \sqrt{0{,}38065} \approx \mathbf{0{,}6170}
    \]
    \[
    \angle \Gamma_{in} = \arctan\left(\frac{-0{,}20860}{-0{,}58064}\right) - 180^\circ = 19{,}76^\circ - 180^\circ = \mathbf{-160{,}24^\circ}
    \]
\end{itemize}
\[
\boxed{\Gamma_{in} = 0{,}617 \angle -160{,}2^\circ}
\]

\subsubsection{(c) Tính toán các độ lợi chuyển đổi và độ lợi toàn chuỗi}
Nguồn phát có trở kháng nội $50~\Omega \implies \Gamma_S = 0$.

\paragraph{1. Độ lợi chuyển đổi của bộ khuếch đại ($G_{T,\text{amp}}$):}
Bộ khuếch đại nối với tải $\Gamma_A = 0{,}0837\angle 90^\circ$ tại mặt phẳng A:
\[
G_{T,\text{amp}} = \frac{|S_{21}|^2 (1 - |\Gamma_S|^2)(1 - |\Gamma_A|^2)}{|1 - S_{22}\Gamma_A|^2 |1 - \Gamma_{in}\Gamma_S|^2}
\]
Thay số với $\Gamma_S = 0$, $|S_{21}| = 4$, $|\Gamma_A| = 0{,}08373$, $|1 - S_{22}\Gamma_A|^2 = 0{,}95376$:
\[
G_{T,\text{amp}} = \frac{4^2 \times 1 \times (1 - 0{,}08373^2)}{0{,}95376 \times 1} = \frac{16 \times 0{,}99299}{0{,}95376} = \frac{15{,}8878}{0{,}95376} \approx \mathbf{16{,}658}
\]
Đổi sang decibel:
\[
G_{T,\text{amp}}(\text{dB}) = 10 \log_{10}(16{,}658) = \mathbf{12{,}22~\text{dB}}
\]

\paragraph{2. Tỉ lệ công suất qua đoạn đường truyền $\lambda/8$ ($G_{T,\text{line}}$):}
Vì đoạn đường truyền là không tổn hao, toàn bộ công suất vào đều truyền qua:
\[
G_{T,\text{line}} \approx 1{,}0 \implies \mathbf{0{,}00~\text{dB}}
\]

\paragraph{3. Tỉ lệ công suất bộ suy giảm chuyển đến tải ($G_{T,\text{att}}$):}
Bộ suy giảm $6~\text{dB}$ phối hợp ($S_{11} = S_{22} = 0$), tải $\Gamma_C = -1/3$:
\[
G_{T,\text{att}} = |S_{21,\text{att}}|^2 (1 - |\Gamma_C|^2) = (0{,}5012)^2 \times \left(1 - \frac{1}{9}\right) = 0{,}2512 \times \frac{8}{9} \approx \mathbf{0{,}2233}
\]
Đổi sang decibel:
\[
G_{T,\text{att}}(\text{dB}) = 10 \log_{10}(0{,}2233) = \mathbf{-6{,}51~\text{dB}}
\]

\paragraph{4. Độ lợi chuyển đổi của cả chuỗi từ nguồn đến tải ($G_{T,\text{total}}$):}
Do các khối nối tiếp nhau, độ lợi chuyển đổi tổng cộng bằng tích các độ lợi từng tầng:
\[
G_{T,\text{total}} = G_{T,\text{amp}} \times G_{T,\text{line}} \times G_{T,\text{att}} = 16{,}658 \times 1{,}0 \times 0{,}2233 \approx \mathbf{3{,}720}
\]
Theo decibel:
\[
G_{T,\text{total}}(\text{dB}) = 12{,}22~\text{dB} + 0{,}00~\text{dB} - 6{,}51~\text{dB} = \mathbf{5{,}71~\text{dB}}
\]
\[
\boxed{G_{T,\text{amp}} = 12{,}22~\text{dB}, \quad G_{T,\text{att}} = -6{,}51~\text{dB}, \quad G_{T,\text{total}} = 5{,}71~\text{dB}}
\]

\subsubsection{(d) Kiểm tra kết quả bằng phương pháp nhân ma trận ABCD}
Ta ghép đoạn dây $\theta = 45^\circ$ và bộ suy giảm $6~\text{dB}$ thành một khối mạng hai cửa duy nhất bằng ma trận truyền đạt $[ABCD]$.
\begin{enumerate}
    \item \textbf{Ma trận ABCD của đoạn đường truyền $50~\Omega, \theta = 45^\circ$:}
    \[
    [ABCD]_{\text{line}} = \begin{bmatrix} \cos\theta & jZ_0\sin\theta \\ jY_0\sin\theta & \cos\theta \end{bmatrix} = \begin{bmatrix} \frac{\sqrt{2}}{2} & j50\frac{\sqrt{2}}{2} \\ j\frac{1}{50}\frac{\sqrt{2}}{2} & \frac{\sqrt{2}}{2} \end{bmatrix} = \begin{bmatrix} 0{,}7071 & j35{,}355 \\ j0{,}01414 & 0{,}7071 \end{bmatrix}
    \]
    \item \textbf{Ma trận ABCD của bộ suy giảm hình T ($R_1 = 16{,}61~\Omega, R_2 = 66{,}93~\Omega$):}
    \[
    [ABCD]_{\text{att}} = \begin{bmatrix} 1 + R_1/R_2 & R_1(2 + R_1/R_2) \\ 1/R_2 & 1 + R_1/R_2 \end{bmatrix} = \begin{bmatrix} 1{,}2482 & 37{,}344 \\ 0{,}01494 & 1{,}2482 \end{bmatrix}
    \]
    \item \textbf{Ma trận ABCD của chuỗi (Đoạn dây + Bộ suy giảm):}
    \[
    [ABCD]_{\text{chain}} = [ABCD]_{\text{line}} \times [ABCD]_{\text{att}}
    \]
    \begin{align*}
    A &= 0{,}7071 \times 1{,}2482 + j35{,}355 \times 0{,}01494 = 0{,}8826 + j0{,}5282 \\
    B &= 0{,}7071 \times 37{,}344 + j35{,}355 \times 1{,}2482 = 26{,}406 + j44{,}130 \\
    C &= j0{,}01414 \times 1{,}2482 + 0{,}7071 \times 0{,}01494 = 0{,}01056 + j0{,}01765 \\
    D &= j0{,}01414 \times 37{,}344 + 0{,}7071 \times 1{,}2482 = 0{,}8826 + j0{,}5282
    \end{align*}
    \item \textbf{Chuyển đổi ma trận $[ABCD]_{\text{chain}}$ sang ma trận $[S]_{\text{chain}}$:}
    Mẫu số chung:
    \[
    \Delta = A + \frac{B}{Z_0} + C Z_0 + D
    \]
    Thay $Z_0 = 50~\Omega$:
    \[
    \frac{B}{50} = 0{,}5281 + j0{,}8826, \qquad 50 C = 0{,}5280 + j0{,}8825
    \]
    \[
    \Delta = (0{,}8826 + j0{,}5282) + (0{,}5281 + j0{,}8826) + (0{,}5280 + j0{,}8825) + (0{,}8826 + j0{,}5282)
    \]
    \[
    \Delta = 2{,}8213 + j2{,}8215 = 3{,}990 \angle 45^\circ
    \]
    Hệ số truyền $S_{21,\text{chain}}$:
    \[
    S_{21,\text{chain}} = \frac{2}{\Delta} = \frac{2}{3{,}990 \angle 45^\circ} = 0{,}5012 \angle -45^\circ
    \]
    Biên độ $|S_{21,\text{chain}}| = 0{,}5012$ tương ứng đúng độ suy giảm $6~\text{dB}$, và góc pha trễ $-45^\circ$ đúng bằng độ dài điện $\theta = 45^\circ$ của đoạn dây.
    
    Hệ số phản xạ $S_{11,\text{chain}}$:
    \[
    S_{11,\text{chain}} = \frac{A + B/Z_0 - C Z_0 - D}{\Delta} = \frac{0{,}5281 + j0{,}8826 - (0{,}5280 + j0{,}8825)}{\Delta} \approx 0
    \]
    Do đó, chuỗi đoạn dây và bộ suy giảm hoạt động tương đương một bộ suy giảm $6~\text{dB}$ phối hợp có độ trễ pha $45^\circ$.
    \item \textbf{Kiểm chứng $\Gamma_A$ và độ lợi toàn phần:}
    Hệ số phản xạ nhìn vào chuỗi khi nối tải $Z_L = 25~\Omega$ ($\Gamma_C = -1/3$):
    \[
    \Gamma_A = S_{11,\text{chain}} + \frac{S_{12,\text{chain}}S_{21,\text{chain}}\Gamma_C}{1 - S_{22,\text{chain}}\Gamma_C} = 0 + (0{,}5012\angle -45^\circ)^2 \times (-1/3)
    \]
    \[
    \Gamma_A = (0{,}2512\angle -90^\circ) \left(\frac{1}{3}\angle 180^\circ\right) = 0{,}0837 \angle 90^\circ = \mathbf{j 0{,}0837}
    \]
    Kết quả này trùng khớp tuyệt đối với kết quả tính từng bước ở câu (a)!
    Độ lợi toàn chuỗi tính qua ma trận tán xạ tương đương cũng cho kết quả:
    \[
    G_{T,\text{total}} = 5{,}71~\text{dB}
    \]
\end{enumerate}
\textbf{Kết luận:} Phương pháp nhân ma trận truyền đạt $ABCD$ cho kết quả hoàn toàn trùng khớp với phương pháp phân tích lan truyền sóng từng khối, khẳng định tính chính xác của toàn bộ bài giải.

\section*{Tổng kết và bài học kinh nghiệm}
\addcontentsline{toc}{section}{Tổng kết và bài học kinh nghiệm}

Báo cáo Bài tập 1 (BT1) đã giải quyết toàn diện 5 bài toán thực tế cốt lõi trong kỹ thuật siêu cao tần:
\begin{enumerate}
    \item \textbf{Bộ suy giảm hình T:} Làm rõ vai trò cải thiện phối hợp trở kháng (matching pad) khi đệm giữa nguồn và tải không chuẩn, với độ cải thiện suy hao phản hồi xấp xỉ gấp đôi độ suy giảm của bộ đệm ($\Delta RL \approx 2L$).
    \item \textbf{Bộ chia Wilkinson:} Khẳng định tính ưu việt của cấu trúc đường truyền $\lambda/4$ kết hợp điện trở cách ly $R = 2Z_0$, đem lại khả năng phối hợp mọi cửa và cách ly hoàn hảo giữa các ngõ ra.
    \item \textbf{Bộ chia điện trở:} Phân tích cấu trúc đơn giản, băng thông rộng nhưng có nhược điểm tiêu hao $50\%$ công suất và độ cách ly kém ($6~\text{dB}$), khiến tải ở một cửa bị lệch sẽ làm biến dạng hệ số truyền của cửa còn lại.
    \item \textbf{Transistor siêu cao tần và tải lệch:} Thể hiện rõ ảnh hưởng của hệ số ghép ngược $S_{12}$ làm thay đổi trở kháng vào khi tải thay đổi, đặt ra yêu cầu phải thiết kế mạch phối hợp trở kháng đồng thời ở hai phía.
    \item \textbf{Ghép chuỗi (Cascade):} Rèn luyện kỹ năng kết hợp phương pháp sóng tán xạ $[S]$ và ma trận truyền đạt $[ABCD]$, kiểm chứng chéo sự bảo toàn năng lượng và độ tin cậy của mô hình toán học.
\end{enumerate}
